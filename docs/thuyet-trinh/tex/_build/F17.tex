\newcommand\stage[8]{%
  \pgfmathsetmacro\xl{(#1-1)*(\bw+\gap)}
  \pgfmathsetmacro\xc{\xl+\bw/2}
  \pgfmathsetmacro\xr{\xl+\bw}
  \filldraw[shad, fill=#2!6, draw=#2!45, line width=1.3pt, rounded corners=12pt] (\xl,\boxbot) rectangle (\xr,\boxtop);
  \begin{scope}
    \clip[rounded corners=12pt] (\xl,\boxbot) rectangle (\xr,\boxtop);
    \fill[#2] (\xl,\boxtop-1.65) rectangle (\xr,\boxtop);
  \end{scope}
  \filldraw[fill=white, draw=white] (\xl+0.95,\boxtop-0.83) circle (0.52);
  \node[#2, font=\bfseries\fontsize{19}{19}\selectfont] at (\xl+0.95,\boxtop-0.83) {#1};
  \node[white, anchor=west, font=\bfseries\fontsize{20}{23}\selectfont] at (\xl+1.68,\boxtop-0.83) {#3};
  \pic[scale=1.3] at (\xc,9.6) {#8};
  \node[cMuc, align=center, text width=\bw cm - 0.5cm] at (\xc,5.9) {#4};

}
